CREATE TABLE pricing_requests (
  id BIGSERIAL PRIMARY KEY,
  customer_id VARCHAR(64) NOT NULL,
  product_type VARCHAR(32) NOT NULL,
  loan_amount DECIMAL(15,2) NOT NULL,
  tenure_months INT NOT NULL,
  credit_score INT NOT NULL,
  annual_income DECIMAL(15,2) NOT NULL,
  calculated_rate DECIMAL(5,2) NOT NULL,
  emi_amount DECIMAL(15,2) NOT NULL,
  trace_id VARCHAR(64),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  version BIGINT NOT NULL -- Optimistic lock
);

Stores every calculated loan quote with zero floating-point loss via DECIMAL(15,2) and standard optimistic locking.
